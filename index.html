function loadCommandCenterData(todayStr) { // 1. โหลดสรุปสถิติพนักงานมาทำงานวันนี้ db.ref('settings/globalLateTime').once('value', (settingsSnap) => { const globalLateTime = settingsSnap.val() || "08:00"; db.ref('employees').once('value', (empSnap) => { const employeesObj = empSnap.val() || {}; let allEmployees = []; Object.keys(employeesObj).forEach(k => allEmployees.push(employeesObj[k])); db.ref('logs').once('value', (logSnap) => { const logsObj = logSnap.val() || {}; let checkedInList = []; Object.keys(logsObj).forEach(k => { if (logsObj[k].date === todayStr && logsObj[k].type === "เข้างาน") { checkedInList.push(logsObj[k]); } }); db.ref('leaves').once('value', (leaveSnap) => { const leavesObj = leaveSnap.val() || {}; let approvedLeavesToday = []; Object.keys(leavesObj).forEach(k => { let l = leavesObj[k]; let statusStr = String(l.status || ''); if (statusStr.includes('อนุมัติแล้ว') && l.startDate <= todayStr && l.endDate >= todayStr) { approvedLeavesToday.push(l); } }); let leaveList = []; let presentList = []; let lateList = []; let absentList = []; allEmployees.forEach(emp => { const present = checkedInList.find(item => String(item.empId) === String(emp.empId)); const leave = approvedLeavesToday.find(item => String(item.empId) === String(emp.empId)); if (leave) { leaveList.push({ emp, leave }); } else if (present) { if (present.time > globalLateTime) { lateList.push({ emp, present }); } else { presentList.push({ emp, present }); } } else { absentList.push({ emp }); } }); let totalPresentCount = presentList.length + lateList.length; // แสดงผลกล่องสรุปซ้ายสุด const statsEl = document.getElementById('ccSummaryStats'); if (statsEl) { statsEl.innerHTML = `
• พนักงานทั้งหมด: ${allEmployees.length} คน
• มาทำงานตรงเวลา: ${presentList.length} คน
• มาสาย: ${lateList.length} คน
• ลางาน: ${leaveList.length} คน
• ยังไม่ลงเวลา: ${absentList.length} คน
`; } // แสดงผลคอลัมน์กลาง: สถานะพนักงานวันนี้ (Live) const attEl = document.getElementById('ccAttendanceList'); if (attEl) { let htmlAtt = ''; leaveList.forEach(item => { htmlAtt += `
🌴 ${item.emp.empName}${item.leave.leaveType}
`; }); presentList.forEach(item => { htmlAtt += `
🟢 ${item.emp.empName}${item.present.time} น.
`; }); lateList.forEach(item => { htmlAtt += `
⏰ ${item.emp.empName}สาย ${item.present.time} น.
`; }); absentList.forEach(item => { htmlAtt += `
🔴 ${item.emp.empName}ยังไม่ลงเวลา
`; }); attEl.innerHTML = htmlAtt || `
ไม่มีข้อมูล
`; } }); }); }); }); // 2. โหลดคอลัมน์ขวาสุด: ติดตามการส่งของ (Live Tracking) db.ref('delivery_jobs/' + todayStr).once('value', (jobSnap) => { const jobsObj = jobSnap.val() || {}; const deliveryEl = document.getElementById('ccDeliveryList'); if (!deliveryEl) return; let jobsList = []; Object.keys(jobsObj).forEach(k => jobsList.push({ key: k, ...jobsObj[k] })); if (jobsList.length === 0) { deliveryEl.innerHTML = `
ยังไม่มีการมอบหมายจ๊อบส่งของในวันนี้
`; return; } let htmlJobs = ''; jobsList.forEach(j => { let isCompleted = String(j.status).includes('สำเร็จ'); let isOnTheWay = String(j.status).includes('กำลังเดินทาง'); let badgeColor = isCompleted ? '#28a745' : (isOnTheWay ? '#0d6efd' : '#e67e22'); let viewPhotoBtn = j.photoUrl ? `🖼️ ดูรูปหลักฐาน` : ''; htmlJobs += `
🚚 คนขับ: ${j.driverName}
🏬 ร้าน: ${j.customerName}
สถานะ: ${j.status} (${j.deliveredTime})
${viewPhotoBtn}
`; }); deliveryEl.innerHTML = htmlJobs; }); }
